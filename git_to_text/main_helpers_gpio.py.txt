"""
GPIO operations helper for KneeSpa.
Handles all GPIO setup, control, and cleanup operations.
"""

import logging
import RPi.GPIO as GPIO
from typing import Dict, Optional, List
from enum import Enum
from config.constants import GPIO_PINS
from utils.logging import setup_logger

class PinState(Enum):
    """GPIO pin states."""
    HIGH = GPIO.HIGH
    LOW = GPIO.LOW

class PinMode(Enum):
    """GPIO pin modes."""
    OUT = GPIO.OUT
    IN = GPIO.IN

class GPIOHelper:
    """
    Helper class for GPIO operations.
    Manages GPIO pin setup, control, and monitoring.
    """

    _instance = None  # Singleton instance

    def __new__(cls):
        """Ensure singleton pattern."""
        if cls._instance is None:
            cls._instance = super(GPIOHelper, cls).__new__(cls)
        return cls._instance

    def __init__(self):
        """Initialize GPIO helper."""
        if hasattr(self, '_initialized'):
            return
        
        self.logger = setup_logger(component='GPIO Helper')
            
        self._initialized = True
        self.pins_in_use: Dict[int, PinMode] = {}
        self.pin_states: Dict[int, PinState] = {}
        
        # Define pin mappings
        self.EMERGENCY_STOP = GPIO_PINS['EMERGENCY_STOP']
        self.EXTRA_FORWARD = GPIO_PINS['EXTRA_FORWARD']
        self.EXTRA_BACKWARD = GPIO_PINS['EXTRA_BACKWARD']
        self.EXTRA_ENABLE = GPIO_PINS['EXTRA_ENABLE']
        
        # Initialize GPIO
        self.setup_gpio()

    def setup_gpio(self) -> None:
        """Initialize GPIO configuration."""
        try:
            GPIO.setmode(GPIO.BCM)
            GPIO.setwarnings(False)

            # Setup pins with their modes
            pin_configs = {
                self.EMERGENCY_STOP: PinMode.OUT,
                self.EXTRA_FORWARD: PinMode.OUT,
                self.EXTRA_BACKWARD: PinMode.OUT,
                self.EXTRA_ENABLE: PinMode.OUT
            }

            for pin, mode in pin_configs.items():
                self.setup_pin(pin, mode)
            
            # Set initial states
            self.write_pin(self.EMERGENCY_STOP, PinState.HIGH)
            self.write_pin(self.EXTRA_ENABLE, PinState.HIGH)
            
            logging.info("GPIO initialization complete")
            
        except Exception as e:
            logging.error(f"GPIO initialization failed: {str(e)}")
            raise

    def setup_pin(self, pin: int, mode: PinMode) -> None:
        """
        Setup individual GPIO pin.
        
        Args:
            pin: GPIO pin number
            mode: Pin mode (IN/OUT)
        """
        try:
            GPIO.setup(pin, mode.value)
            self.pins_in_use[pin] = mode
            logging.debug(f"Pin {pin} setup as {mode.name}")
        except Exception as e:
            logging.error(f"Failed to setup pin {pin}: {str(e)}")
            raise

    def write_pin(self, pin: int, state: PinState) -> None:
        """
        Set GPIO pin state.
        
        Args:
            pin: GPIO pin number
            state: Desired pin state (HIGH/LOW)
        """
        try:
            GPIO.output(pin, state.value)
            self.pin_states[pin] = state
            logging.debug(f"Pin {pin} set to {state.name}")
        except Exception as e:
            logging.error(f"Failed to write to pin {pin}: {str(e)}")
            raise

    def read_pin(self, pin: int) -> Optional[PinState]:
        """
        Read GPIO pin state.
        
        Args:
            pin: GPIO pin number
            
        Returns:
            Optional[PinState]: Current pin state if successful
        """
        try:
            value = GPIO.input(pin)
            return PinState.HIGH if value else PinState.LOW
        except Exception as e:
            logging.error(f"Failed to read pin {pin}: {str(e)}")
            return None

    def cleanup(self) -> None:
        """Clean up GPIO resources."""
        try:
            # Reset all pins to safe state before cleanup
            for pin in self.pins_in_use:
                self.write_pin(pin, PinState.LOW)
            GPIO.cleanup()
            logging.info("GPIO cleanup complete")
        except Exception as e:
            logging.error(f"GPIO cleanup failed: {str(e)}")

    # Movement Control Methods
    def forward_movement(self) -> None:
        """Configure pins for forward movement."""
        try:
            self.write_pin(self.EXTRA_BACKWARD, PinState.LOW)
            self.write_pin(self.EXTRA_FORWARD, PinState.HIGH)
            logging.debug("Forward movement activated")
        except Exception as e:
            logging.error(f"Forward movement failed: {str(e)}")
            self.emergency_stop()

    def reverse_movement(self) -> None:
        """Configure pins for reverse movement."""
        try:
            self.write_pin(self.EXTRA_FORWARD, PinState.LOW)
            self.write_pin(self.EXTRA_BACKWARD, PinState.HIGH)
            logging.debug("Reverse movement activated")
        except Exception as e:
            logging.error(f"Reverse movement failed: {str(e)}")
            self.emergency_stop()

    def stop_movement(self) -> None:
        """Stop all movement."""
        try:
            self.write_pin(self.EXTRA_FORWARD, PinState.LOW)
            self.write_pin(self.EXTRA_BACKWARD, PinState.LOW)
            logging.debug("Movement stopped")
        except Exception as e:
            logging.error(f"Stop movement failed: {str(e)}")
            self.emergency_stop()

    def emergency_stop(self) -> None:
        """Trigger emergency stop."""
        try:
            # Set all movement pins to LOW
            self.write_pin(self.EXTRA_FORWARD, PinState.LOW)
            self.write_pin(self.EXTRA_BACKWARD, PinState.LOW)
            self.write_pin(self.EMERGENCY_STOP, PinState.LOW)
            logging.warning("Emergency stop triggered")
        except Exception as e:
            logging.critical(f"Emergency stop failed: {str(e)}")
            # As a last resort, try direct GPIO cleanup
            GPIO.cleanup()

    def reset_system(self) -> bool:
        """
        Reset system to safe state.
        
        Returns:
            bool: True if reset successful
        """
        try:
            # Reset all pins to default states
            self.write_pin(self.EMERGENCY_STOP, PinState.HIGH)
            self.write_pin(self.EXTRA_ENABLE, PinState.HIGH)
            self.write_pin(self.EXTRA_FORWARD, PinState.LOW)
            self.write_pin(self.EXTRA_BACKWARD, PinState.LOW)
            
            logging.info("System reset complete")
            return True
            
        except Exception as e:
            logging.error(f"System reset failed: {str(e)}")
            self.emergency_stop()
            return False

    def get_pin_status(self) -> Dict[str, Dict[str, str]]:
        """
        Get current status of all pins.
        
        Returns:
            Dict containing current pin configurations and states
        """
        status = {}
        for pin in self.pins_in_use:
            status[str(pin)] = {
                'mode': self.pins_in_use[pin].name,
                'state': self.pin_states.get(pin, PinState.LOW).name
            }
        return status

# Global GPIO instance
gpio = GPIOHelper()
